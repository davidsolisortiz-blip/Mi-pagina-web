    <script type="text/javascript">
        function sumar() {
            let num1 = parseFloat(document.getElementById('numero1').value);
            let num2 = parseFloat(document.getElementById('numero2').value);

            if (isNaN(num1) || isNaN(num2)) {
                document.getElementById('resultado').textContent = "⚠️ INGRESA AMBOS VALORES";
                return;
            }

            let suma = num1 + num2;
            document.getElementById('resultado').textContent = "PODER TOTAL = " + suma;
        }
    </script>
